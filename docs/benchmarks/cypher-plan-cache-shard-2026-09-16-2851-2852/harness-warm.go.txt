package cypher

// zz_2852_warm_test.go — THROWAWAY measurement harness for rmp #2852.
//
// It is the harness archived at
// docs/benchmarks/cypher-parse-lab-2026-09-16-spike2846/harness-cypher-warm.go.txt,
// restored VERBATIM (only the file name and this header differ) so the A arm of
// this task's A/B is measured by exactly the code that measured the finding.
// The benchmark names are unchanged, so benchstat can also place the new series
// beside the archived warm.series.txt.
//
// Question: does the warm statement-preparation path — StripLiterals plus the
// plan-cache lookup, which is everything [Engine.parseAndAnalyse] does on a hit
// — scale with the cores given to it, or does planCache's mutex serialise every
// caller?
//
// Method: [Engine.parseAndAnalyse] — the production entry point, reached
// verbatim by [Engine.runRead] — on a pre-warmed cache, under b.RunParallel at
// -cpu=1,2,4,8,10. A path that scales reports a flat ns/op as -cpu rises; a
// path that serialises reports a rising one.
//
// Two shapes, because the strip's cost differs by two orders between them:
//   quotefree — StripLiterals answers in one IndexByte and returns ok=false
//   literal   — StripLiterals scans, folds every identifier and rewrites

import (
	"fmt"
	"testing"

	"github.com/FlavioCFOliveira/GoGraph/cypher/parser"
)

// parserStripForSpike names the production strip so the key this file checks is
// resolved exactly as [Engine.parseAndAnalyse] resolves it.
func parserStripForSpike(q string) (string, map[string]string, bool) {
	return parser.StripLiterals(q)
}

// BenchmarkSpikeWarmPrepareParallel measures the warm prepare path under
// concurrency. One op = one parseAndAnalyse on a warm cache.
func BenchmarkSpikeWarmPrepareParallel(b *testing.B) {
	shapes := []struct {
		name string
		text string
	}{
		{"quotefree", "MATCH (u:USER {id:$id}) RETURN u.age AS c"},
		{"literal", "MATCH (u:USER) WHERE u.city = 'city-3' RETURN count(u) AS c"},
	}
	for _, s := range shapes {
		b.Run(s.name, func(b *testing.B) {
			eng := newShareRig(b)
			// Publish the entry, and prove it is published: a benchmark that
			// silently measured a miss would answer a different question.
			if _, _, err := eng.parseAndAnalyse(s.text); err != nil {
				b.Fatalf("warm: %v", err)
			}
			key := s.text
			if stripped, _, ok := parserStripForSpike(s.text); ok {
				key = stripped
			}
			if _, ok := eng.cache.get(key); !ok {
				b.Fatalf("cache is not warm for %q", s.text)
			}
			b.ReportAllocs()
			b.ResetTimer()
			b.RunParallel(func(pb *testing.PB) {
				for pb.Next() {
					if _, _, err := eng.parseAndAnalyse(s.text); err != nil {
						b.Fatal(err)
					}
				}
			})
		})
	}
}

// BenchmarkSpikeWarmPrepareMultiKey answers the obvious objection to the
// single-key benchmark above: that its contention is an artefact of every
// goroutine touching the same LRU node. Here each goroutine walks a distinct
// set of 32 warm keys, so no two concurrent lookups share a map entry or an
// LRU position.
func BenchmarkSpikeWarmPrepareMultiKey(b *testing.B) {
	const nKeys = 32
	eng := newShareRig(b)
	texts := make([]string, nKeys)
	for i := range texts {
		texts[i] = fmt.Sprintf("MATCH (u:USER {id:$id}) RETURN u.age AS c%d", i)
		if _, _, err := eng.parseAndAnalyse(texts[i]); err != nil {
			b.Fatalf("warm %d: %v", i, err)
		}
		if _, ok := eng.cache.get(texts[i]); !ok {
			b.Fatalf("cache is not warm for key %d", i)
		}
	}
	b.ReportAllocs()
	b.ResetTimer()
	b.RunParallel(func(pb *testing.PB) {
		i := 0
		for pb.Next() {
			if _, _, err := eng.parseAndAnalyse(texts[i%nKeys]); err != nil {
				b.Fatal(err)
			}
			i++
		}
	})
}
